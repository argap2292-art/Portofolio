💡 Petunjuk Hak Akses (Row Level Security / RLS) di Supabase:

Jika Anda mengalami pesan eror new row violates row-level security policy saat menyimpan data, pastikan Anda telah mengaktifkan Policy akses publik di Supabase SQL Editor dengan menjalankan perintah berikut: 

-- 1. Tambahkan kolom yang belum ada (jika belum dibuat)
ALTER TABLE "Publikasi" ADD COLUMN IF NOT EXISTS "Nomor ISBN/DOI" text, ADD COLUMN IF NOT EXISTS "Abstrak" text;
ALTER TABLE "Karya Inovasi" ADD COLUMN IF NOT EXISTS "Deskripsi Lengkap" text, ADD COLUMN IF NOT EXISTS "Tautan Eksternal/Dok" text;

-- 2. Izinkan akses baca dan tulis untuk anonim (RLS)
ALTER TABLE "Publikasi" ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Akses Publik Publikasi" ON "Publikasi";
CREATE POLICY "Akses Publik Publikasi" ON "Publikasi" FOR ALL TO anon USING (true) WITH CHECK (true);

ALTER TABLE "Karya Inovasi" ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Akses Publik Karya Inovasi" ON "Karya Inovasi";
CREATE POLICY "Akses Publik Karya Inovasi" ON "Karya Inovasi" FOR ALL TO anon USING (true) WITH CHECK (true);

-- 3. Konfigurasi Bucket Storage "berkas-publikasi" untuk File PDF
INSERT INTO storage.buckets (id, name, public) VALUES ('berkas-publikasi', 'berkas-publikasi', true)
ON CONFLICT (id) DO UPDATE SET public = true;

DROP POLICY IF EXISTS "Akses Publik Berkas Publikasi" ON storage.objects;
CREATE POLICY "Akses Publik Berkas Publikasi" ON storage.objects FOR SELECT TO anon USING (bucket_id = 'berkas-publikasi');

DROP POLICY IF EXISTS "Upload Publik Berkas Publikasi" ON storage.objects;
CREATE POLICY "Upload Publik Berkas Publikasi" ON storage.objects FOR INSERT TO anon WITH CHECK (bucket_id = 'berkas-publikasi');

-- 4. Konfigurasi Bucket Storage "berkas-inovasi" untuk Dokumen/Media/Foto
INSERT INTO storage.buckets (id, name, public) VALUES ('berkas-inovasi', 'berkas-inovasi', true)
ON CONFLICT (id) DO UPDATE SET public = true;

DROP POLICY IF EXISTS "Akses Publik Berkas Inovasi" ON storage.objects;
CREATE POLICY "Akses Publik Berkas Inovasi" ON storage.objects FOR SELECT TO anon USING (bucket_id = 'berkas-inovasi');

DROP POLICY IF EXISTS "Upload Publik Berkas Inovasi" ON storage.objects;
CREATE POLICY "Upload Publik Berkas Inovasi" ON storage.objects FOR INSERT TO anon WITH CHECK (bucket_id = 'berkas-inovasi');
